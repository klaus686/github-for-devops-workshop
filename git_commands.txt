 git init                                                         =   to setup VCS in a folder
 git status                                                       =   to check the staus of git 
 git restore --staged <filename>                                        =   to remove file from stage
 git add                                                          =   to add file on stage
 git commit -m "commit message"                                   =   to add file in VCS
 git restore <filename>                                             =   to restore deleted file
 git remote -v                                                    =   to check the remote repo
 git remote set-url origin https://PAT@github.com/usernane/repo   =   to set remote access to that folder so we can push file or folder to that repo
 git push origin master                                           =   to push the files and folder to github from local
 git pull origin master                                           =   to pull the files and folder from github to local
 git branch                                                       =   to check all branches of a repository
 git branch <newbranchname>                                       =   to create a new branch
 git switch <branchname>                                          =   to switch to a branch
 git log                                                          =   to check the commits with all details
 git log --oneline                                                =   to check commits with commit message only
 git checkout <branchname>                                        =   also switch branch
